"""api/kb_project.py — 单项目知识库的完整封装。"""

import os, json, time, hashlib, shutil, zipfile
from typing import List, Dict, Any, Optional

class KBProject:
    """封装单个项目的所有操作：文档、切片、向量、检索、备份、配置。"""

    def __init__(self, project_dir: str, project_config: dict = None):
        self.project_dir = project_dir
        os.makedirs(project_dir, exist_ok=True)

        self.config_path = os.path.join(project_dir, "config.json")
        self.folders_path = os.path.join(project_dir, "folders.json")
        self.documents_path = os.path.join(project_dir, "documents.json")
        self.chunks_path = os.path.join(project_dir, "chunks.json")
        self.vectors_path = os.path.join(project_dir, "vectors.npy")
        self.raw_docs_dir = os.path.join(project_dir, "raw_docs")
        self.backups_dir = os.path.join(project_dir, "backups")
        os.makedirs(self.raw_docs_dir, exist_ok=True)
        os.makedirs(self.backups_dir, exist_ok=True)

        # 默认配置
        self.config = {
            "chunk_size_min": 100,
            "chunk_size_max": 500,
            "embedding_model": "bge-small-zh",
            "custom_vocab": [],
        }
        self.folders: List[Dict] = []
        self.documents: List[Dict] = []
        self._chunks: List[Dict] = []
        self._vectors = None
        self._model = None
        self._hash_vectorizer = None
        self._embedding_backend = "sentence_transformers"
        self._bm25 = None

        if project_config:
            self.config.update(project_config)

        self._load_state()

    def _load_state(self):
        """从磁盘加载项目状态。"""
        import numpy as np
        if os.path.exists(self.config_path):
            with open(self.config_path, "r", encoding="utf-8") as f:
                self.config.update(json.load(f))
        if os.path.exists(self.folders_path):
            with open(self.folders_path, "r", encoding="utf-8") as f:
                self.folders = json.load(f)
        if os.path.exists(self.documents_path):
            with open(self.documents_path, "r", encoding="utf-8") as f:
                self.documents = json.load(f)
        if os.path.exists(self.chunks_path):
            with open(self.chunks_path, "r", encoding="utf-8") as f:
                data = json.load(f)
                self._chunks = data if isinstance(data, list) else data.get("chunks", [])
        if os.path.exists(self.vectors_path):
            try:
                self._vectors = np.load(self.vectors_path)
            except Exception:
                self._vectors = None
        self._apply_custom_vocab()
        self._rebuild_bm25()

    def _save_state(self):
        """保存所有状态到磁盘。"""
        import numpy as np
        with open(self.config_path, "w", encoding="utf-8") as f:
            json.dump(self.config, f, ensure_ascii=False, indent=2)
        with open(self.folders_path, "w", encoding="utf-8") as f:
            json.dump(self.folders, f, ensure_ascii=False, indent=2)
        with open(self.documents_path, "w", encoding="utf-8") as f:
            json.dump(self.documents, f, ensure_ascii=False, indent=2)
        with open(self.chunks_path, "w", encoding="utf-8") as f:
            json.dump(self._chunks, f, ensure_ascii=False, indent=2)
        if self._vectors is not None:
            np.save(self.vectors_path, self._vectors)

    def get_file_hash(self, file_path: str) -> str:
        hasher = hashlib.md5()
        with open(file_path, "rb") as f:
            for chunk in iter(lambda: f.read(1024 * 1024), b""):
                hasher.update(chunk)
        return hasher.hexdigest()

    def add_document(self, file_path: str, filename: str, content: str,
                     doc_type: str = "unknown", file_size: int = 0,
                     folder_id: str = "", note: str = "") -> Dict:
        """添加文档并自动切片向量化。"""
        import numpy as np
        file_hash = self.get_file_hash(file_path)

        # 检查是否已存在同名同 hash → 覆盖更新
        existing = self.get_doc_by_hash(file_hash)
        if existing:
            return self._update_document(file_hash, file_path, content, doc_type, file_size, folder_id, note)

        # 处理文件名冲突（同名不同内容）
        doc_id = file_hash
        base_name, ext = os.path.splitext(filename)
        conflicting = [d for d in self.documents if d.get("filename") == filename]
        if conflicting:
            filename = f"{base_name} ({len(conflicting)}){ext}"

        # 复制原始文件到 raw_docs
        raw_path = os.path.join(self.raw_docs_dir, f"{file_hash}{ext.lower()}")
        shutil.copy2(file_path, raw_path)

        doc = {
            "id": doc_id,
            "file_hash": file_hash,
            "filename": filename,
            "doc_type": doc_type,
            "file_size": file_size,
            "folder_id": folder_id or "",
            "note": note,
            "added_at": int(time.time()),
            "updated_at": int(time.time()),
            "chunk_count": 0,
            "vector_status": "pending",
        }

        # 切片 + 向量化
        chunks = self._chunk_document(content)
        if chunks:
            vectors = self._encode_texts([c["content"] for c in chunks])
            new_records = []
            for i, ch in enumerate(chunks):
                new_records.append({
                    "id": f"{file_hash}_{i}",
                    "doc_id": doc_id,
                    "content": ch["content"],
                    "metadata": {
                        "filename": filename,
                        "doc_id": doc_id,
                        "folder_id": folder_id or "",
                        "section_title": ch.get("section_title", ""),
                        "chunk_index": i,
                        "added_at": int(time.time()),
                    },
                })

            if self._vectors is None or len(self._chunks) == 0:
                self._vectors = vectors
                self._chunks = new_records
            else:
                self._vectors = np.vstack([self._vectors, vectors])
                self._chunks.extend(new_records)

            doc["chunk_count"] = len(new_records)
            doc["vector_status"] = "vectorized"

        self.documents.insert(0, doc)
        self._rebuild_bm25()
        self._save_state()
        return {"success": True, "doc_id": doc_id, "chunks": len(new_records or [])}

    def _update_document(self, file_hash: str, file_path: str, content: str,
                         doc_type: str, file_size: int, folder_id: str, note: str) -> Dict:
        """覆盖更新已有文档（保留备注和文件夹归属）。"""
        existing = self.get_doc_by_hash(file_hash)
        if not existing:
            return {"success": False, "message": "文档不存在"}

        # 删除旧向量
        self._remove_doc_vectors(file_hash)

        # 更新元数据
        ext = os.path.splitext(existing["filename"])[1]
        raw_path = os.path.join(self.raw_docs_dir, f"{file_hash}{ext}")
        shutil.copy2(file_path, raw_path)

        existing["file_size"] = file_size
        existing["doc_type"] = doc_type
        existing["updated_at"] = int(time.time())
        if folder_id:
            existing["folder_id"] = folder_id
        if note:
            existing["note"] = note

        # 重新切片向量化
        chunks = self._chunk_document(content)
        if chunks:
            vectors = self._encode_texts([c["content"] for c in chunks])
            new_records = []
            for i, ch in enumerate(chunks):

                    "content": ch["content"],
                    "metadata": {
                        "filename": existing["filename"],
                        "doc_id": existing["id"],
                        "folder_id": existing["folder_id"],
                        "section_title": ch.get("section_title", ""),
                        "chunk_index": i,
                        "added_at": int(time.time()),
                    },
                })
            if self._vectors is None:
                self._vectors = vectors
            else:
                self._vectors = np.vstack([self._vectors, vectors])
            self._chunks.extend(new_records)
            existing["chunk_count"] = len(new_records)
            existing["vector_status"] = "vectorized"

        self._rebuild_bm25()
        self._save_state()
        return {"success": True, "doc_id": existing["id"], "chunks": len(new_records)}

    # ==== 智能切片 ====

    def _chunk_document(self, content: str) -> List[Dict]:
        """标题感知切片 + token 回退。"""
        import re, jieba

        clean = re.sub(r'<[^>]+>', ' ', content)
        clean = re.sub(r'\s+', ' ', clean).strip()
        if not clean:
            return []

        cmin = self.config.get("chunk_size_min", 100)
        cmax = self.config.get("chunk_size_max", 500)
        overlap_ratio = 0.1

        # Step 1: 尝试按 Markdown 标题分割
        heading_pattern = re.compile(r'^(#{1,3})\s+(.+)$', re.MULTILINE)
        heading_matches = list(heading_pattern.finditer(clean))

        if heading_matches:
            return self._chunk_by_headings(clean, heading_matches, cmin, cmax)
        else:
            # Step 2: 尝试按段落分割
            paragraphs = re.split(r'\n\s*\n', clean)
            if len(paragraphs) > 1 and all(len(p) < cmax * 2 for p in paragraphs if p.strip()):
                return self._chunk_by_paragraphs(paragraphs, cmin, cmax)
            else:
                # Step 3: 回退到 token 切分
                return self._chunk_by_tokens(clean, cmin, cmax, overlap_ratio)

    def _chunk_by_headings(self, text: str, heading_matches: list,
                           cmin: int, cmax: int) -> List[Dict]:
        """按 Markdown 标题分割文本块。"""
        chunks = []
        for i, match in enumerate(heading_matches):
            start = match.end()
            end = heading_matches[i + 1].start() if i + 1 < len(heading_matches) else len(text)
            section_text = text[start:end].strip()
            if not section_text:
                continue

            title = match.group(2).strip()
            level = len(match.group(1))

            # 如果章节过长，按段落二次切分
            if len(section_text) > cmax:
                sub_paras = re.split(r'\n\s*\n', section_text)
                current = []
                current_len = 0
                for para in sub_paras:
                    para = para.strip()
                    if not para:
                        continue
                    if current_len + len(para) > cmax and current:
                        chunks.append({
                            "content": "\n".join(current),
                            "section_title": title,
                        })
                        current = [para]
                        current_len = len(para)
                    else:
                        current.append(para)
                        current_len += len(para)
                if current:
                    chunks.append({
                        "content": "\n".join(current),
                        "section_title": title,
                    })
            else:
                if len(section_text) >= cmin:
                    chunks.append({
                        "content": section_text,
                        "section_title": title,
                    })
        return chunks

    def _chunk_by_paragraphs(self, paragraphs: List[str], cmin: int, cmax: int) -> List[Dict]:
        """按段落合并切分。"""
        chunks = []
        current = []
        current_len = 0
        for para in paragraphs:
            para = para.strip()
            if not para:
                continue
            if current_len + len(para) > cmax and current:
                chunks.append({"content": "\n".join(current), "section_title": ""})
                current = [para]
                current_len = len(para)
            else:
                current.append(para)
                current_len += len(para)
        if current:
            combined = "\n".join(current)
            if len(combined) >= cmin:
                chunks.append({"content": combined, "section_title": ""})
        return chunks

    def _chunk_by_tokens(self, text: str, cmin: int, cmax: int,
                         overlap_ratio: float) -> List[Dict]:
        """按 token 数切分（回退方案）。"""
        import jieba, random
        chunk_tokens = random.randint(cmin, cmax)
        tokens = [t for t in jieba.lcut(text) if t.strip()]
        if not tokens:
            return []
        overlap = max(1, int(chunk_tokens * overlap_ratio))
        chunks = []
        start = 0
        while start < len(tokens):
            end = min(len(tokens), start + chunk_tokens)
            chunk_text = "".join(tokens[start:end])
            chunks.append({"content": chunk_text, "section_title": ""})
            if end >= len(tokens):
                break
            start = max(0, end - overlap)
        return chunks

    # ==== 向量化 ====

    def _ensure_model(self):
        if self._model is not None:
            return
        model_name = self.config.get("embedding_model", "bge-small-zh")
        try:
            from sentence_transformers import SentenceTransformer
            hf_home = os.path.join(os.path.dirname(self.project_dir), "hf_cache")
            os.makedirs(hf_home, exist_ok=True)
            os.environ.setdefault("HF_HOME", hf_home)
            os.environ.setdefault("TRANSFORMERS_CACHE", os.path.join(hf_home, "transformers"))
            os.environ.setdefault("SENTENCE_TRANSFORMERS_HOME", os.path.join(hf_home, "sentence_transformers"))
            os.environ.setdefault("TOKENIZERS_PARALLELISM", "false")

            model_map = {
                "bge-small-zh": "BAAI/bge-small-zh-v1.5",
                "bge-large-zh": "BAAI/bge-large-zh-v1.5",
                "text2vec-base": "shibing624/text2vec-base-chinese",
            }
            hf_model = model_map.get(model_name, "BAAI/bge-small-zh-v1.5")
            self._model = SentenceTransformer(hf_model)
            self._embedding_backend = "sentence_transformers"
        except Exception as e:
            from sklearn.feature_extraction.text import HashingVectorizer
            self._model = None
            self._embedding_backend = "hashing"
            self._hash_vectorizer = HashingVectorizer(n_features=2048, alternate_sign=False, norm=None)

    def _encode_texts(self, texts: List[str]):
        import numpy as np
        from sklearn.preprocessing import normalize
        self._ensure_model()
        if self._embedding_backend == "sentence_transformers" and self._model is not None:
            vec = self._model.encode(texts, normalize_embeddings=True)
            return np.asarray(vec, dtype=np.float32)
        if self._hash_vectorizer is None:
            from sklearn.feature_extraction.text import HashingVectorizer
            self._hash_vectorizer = HashingVectorizer(n_features=2048, alternate_sign=False, norm=None)
        sparse = self._hash_vectorizer.transform(texts)
        sparse = normalize(sparse, norm="l2", copy=False)
        return sparse.astype(np.float32).toarray()

    def _rebuild_bm25(self):
        import jieba
        from rank_bm25 import BM25Okapi
        if not self._chunks:
            self._bm25 = None
            return
        tokenized = [jieba.lcut(c.get("content", "")) for c in self._chunks]

    # ==== 检索 ====

    def search(self, query: str, top_k: int = 5, folder_id: str = None) -> List[Dict]:
        """混合检索：向量语义 + BM25 关键词 + RRF 融合。"""
        import numpy as np
        import jieba
        if not self._chunks:
            return []

        # 过滤 chunks（按文件夹）
        chunk_indices = range(len(self._chunks))
        if folder_id:
            chunk_indices = [i for i in range(len(self._chunks))
                            if self._chunks[i].get("metadata", {}).get("folder_id") == folder_id]
        if not chunk_indices:
            return []

        q_vec = self._encode_texts([query])
        # 向量检索
        vec_scores = None
        if self._vectors is not None and self._vectors.shape[0] == len(self._chunks):
            subset_vec = self._vectors[list(chunk_indices)]
            vec_scores = (subset_vec @ q_vec[0]).astype(np.float32)
            top_n = min(len(chunk_indices), max(top_k * 2, 10))
            sorted_idx = np.argsort(-vec_scores)[:top_n]
            scored = [(float(vec_scores[i]), self._chunks[chunk_indices[i]]) for i in sorted_idx]
        else:
            scored = [(0.0, self._chunks[i]) for i in chunk_indices[:top_k]]

        # BM25（如果在全量索引上）
        bm25_scored = {}
        if self._bm25 is not None and len(self._chunks) > 0:
            tokenized_query = jieba.lcut(query)
            scores = self._bm25.get_scores(tokenized_query)
            top_n = min(len(chunk_indices), max(top_k * 2, 10))
            idxs = np.argsort(-scores)[:top_n]
            for rank, idx in enumerate(idxs):
                if scores[idx] <= 0:
                    continue
                bm25_scored[self._chunks[idx]["id"]] = 1.0 / (rank + 1)

        # RRF 融合
        combined = {}
        for vec_score, chunk in scored:
            cid = chunk["id"]
            combined[cid] = vec_score + bm25_scored.get(cid, 0)

        sorted_chunks = sorted(scored, key=lambda x: combined.get(x[1]["id"], x[0]), reverse=True)[:top_k]
        return [
            {"content": c["content"], "metadata": c["metadata"],
             "score": round(float(combined.get(c["id"], s)), 4)}
            for s, c in sorted_chunks
        ]

    def search_by_categories(self, query: str, categories: List[str] = None,
                              top_k_per_category: int = 3) -> Dict[str, List[Dict]]:
        """按文件夹分类检索（兼容旧接口）。"""
        if not categories:
            categories = [f["name"] for f in self.folders]
        if not categories:
            return {"通用": self.search(query, top_k=top_k_per_category)}

        result = {}
        for cat in categories:
            folder = next((f for f in self.folders if f["name"] == cat), None)
            if folder:
                result[cat] = self.search(query, top_k=top_k_per_category, folder_id=folder["id"])
            else:
                result[cat] = []
        return result

    def fuzzy_search(self, keyword: str, folder_id: str = None) -> List[Dict]:
        """关键词模糊匹配（基于 jieba 分词 + 子串匹配）。"""
        import jieba
        results = []
        tokens = set(jieba.lcut_for_search(keyword))
        for chunk in self._chunks:
            if folder_id and chunk.get("metadata", {}).get("folder_id") != folder_id:
                continue
            content = chunk.get("content", "")
            # 检查是否包含关键词或分词后的任意 token
            if keyword.lower() in content.lower():
                results.append(chunk)
            else:
                content_tokens = set(jieba.lcut_for_search(content))
                if tokens & content_tokens:
                    results.append(chunk)
        # 去重
        seen = set()
        unique = []
        for r in results:
            if r["id"] not in seen:
                seen.add(r["id"])
                unique.append(r)
        return [{"content": c["content"], "metadata": c["metadata"], "score": 0.0}
                for c in unique[:20]]

@router.post("/project/{project_id}/upload")
async def upload_document(project_id: str, request: Request):
    """上传文档（单文件 multipart 或二进制流）。"""
    proj = get_kb().get_project(project_id)
    if not proj:
        raise HTTPException(status_code=404, detail="项目不存在")

    filename = request.headers.get("X-Filename", "").strip()
    if filename:
        try:
            filename = unquote(filename)
        except Exception:
            pass
    folder_id = request.headers.get("X-Folder-Id", "")
    note = request.headers.get("X-Note", "")

    if not filename:
        # 尝试从 multipart 读取
        form = await request.form()
        file = form.get("file")
        if not file:
            raise HTTPException(status_code=400, detail="缺少文件")
        filename = file.filename or "unknown"
        folder_id = form.get("folder_id", "")
        note = form.get("note", "")
        raw_bytes = await file.read()
    else:
        raw_bytes = await request.body()

    if not raw_bytes or len(raw_bytes) < 20:
        raise HTTPException(status_code=400, detail="文件内容为空")

    safe_filename = os.path.basename(filename)
    ext = os.path.splitext(safe_filename)[1].lower()
    file_id = str(uuid.uuid4())
    save_name = f"doc_{file_id}{ext}"
    file_path = os.path.join(_upload_dir(), save_name)
    with open(file_path, "wb") as f:
        f.write(raw_bytes)

    content = DocumentParser.parse(file_path)
    if isinstance(content, str) and content.startswith("DOCX 解析错误"):
        return {"success": False, "message": content}

    doc_type = ext.lstrip(".") or "unknown"
    result = proj.add_document(
        file_path=file_path, filename=safe_filename,
        content=str(content), doc_type=doc_type,
        file_size=len(raw_bytes), folder_id=folder_id, note=note,
    )
    if os.path.exists(file_path):
        os.remove(file_path)
    return result


@router.post("/project/{project_id}/upload-multi")
async def upload_multiple(project_id: str, request: Request):
    """批量上传文档（multipart 多文件）。"""
    proj = get_kb().get_project(project_id)
    if not proj:
        raise HTTPException(status_code=404, detail="项目不存在")

    form = await request.form()
    results = []
    for key in form.keys():
        field = form.get(key)
        if hasattr(field, "filename") and field.filename:
            raw_bytes = await field.read()
            if not raw_bytes or len(raw_bytes) < 20:
                continue
            safe_name = os.path.basename(field.filename)
            ext = os.path.splitext(safe_name)[1].lower()
            if ext not in SUPPORTED_EXTS and ext not in {'.csv'}:
                continue
            file_id = str(uuid.uuid4())
            file_path = os.path.join(_upload_dir(), f"doc_{file_id}{ext}")
            with open(file_path, "wb") as f:
                f.write(raw_bytes)
            content = DocumentParser.parse(file_path)
            doc_type = ext.lstrip(".") or "unknown"
            result = proj.add_document(
                file_path=file_path, filename=safe_name,
                content=str(content) if content else "",
                doc_type=doc_type, file_size=len(raw_bytes),
            )
            results.append({"filename": safe_name, **result})
            if os.path.exists(file_path):
                os.remove(file_path)
    return {"success": True, "data": {"results": results, "total": len(results)}}

@router.post("/project/{project_id}/doc/{doc_id}/revectorize")
async def revectorize_document(project_id: str, doc_id: str):
    proj = get_kb().get_project(project_id)
    if not proj:
        raise HTTPException(status_code=404, detail="项目不存在")
    # 先删除旧向量，重新添加
    proj._remove_doc_vectors(doc_id)
    # 找到原始文件重新处理
    doc = next((d for d in proj.documents if d["id"] == doc_id), None)
    if not doc:
        raise HTTPException(status_code=404, detail="文档不存在")
    file_hash = doc.get("file_hash", doc_id)
    ext = "." + doc.get("doc_type", "txt")
    raw_path = os.path.join(proj.raw_docs_dir, f"{file_hash}{ext}")
    if not os.path.exists(raw_path):
        raw_path = os.path.join(proj.raw_docs_dir, doc["filename"])
    if not os.path.exists(raw_path):
        return {"success": False, "message": "原始文件不存在，无法重向量化"}
    from document_parser import DocumentParser
    content = DocumentParser.parse(raw_path)
    if not content:
        return {"success": False, "message": "解析失败"}
    result = proj.add_document(raw_path, doc["filename"], str(content),
                                doc["doc_type"], doc["file_size"],
                                doc.get("folder_id", ""), doc.get("note", ""))
    return result

@router.post("/global-search")
async def global_search(payload: dict = Body(...)):
    """跨所有未归档项目检索。"""
    query = payload.get("query", "").strip()
    if not query:
        raise HTTPException(status_code=400, detail="查询内容不能为空")
    top_k = int(payload.get("top_k", 3))
    results = []
    for pinfo in get_kb()._projects:
        if pinfo.get("archived"):
            continue
        proj = get_kb().get_project(pinfo["id"])
        if not proj:
            continue
        res = proj.search(query, top_k=top_k)
        for r in res:
            r["project_name"] = pinfo["name"]
            r["project_id"] = pinfo["id"]
        results.extend(res)
    results.sort(key=lambda x: x.get("score", 0), reverse=True)
    return {"success": True, "data": {"results": results, "total": len(results)}}
